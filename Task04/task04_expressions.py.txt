import math
import time

# -----------------------------
# Rate card constants
# -----------------------------
BASE_RATE = 30.00
BASE_FAT = 3.5
BASE_SNF = 8.5

STEP_SIZE = 0.1

FAT_STEP_VALUE = 0.40
SNF_STEP_VALUE = 0.30

MIN_FAT = 2.0
MAX_FAT = 7.0


def fat_steps_round(fat):
    """Return the number of fat steps using round()."""
    return round((fat - BASE_FAT) / STEP_SIZE)


def snf_steps_round(snf):
    """Return the number of SNF steps using round()."""
    return round((snf - BASE_SNF) / STEP_SIZE)


def rate_per_litre(fat, snf):
    """Return the correct rate per litre, rounded to 2 decimals."""
    rate = (BASE_RATE
            + fat_steps_round(fat) * FAT_STEP_VALUE
            + snf_steps_round(snf) * SNF_STEP_VALUE)
    return round(rate, 2)


def payout(litres, fat, snf):
    """Return the amount payable for one entry, rounded to 2 decimals."""
    return round(litres * rate_per_litre(fat, snf), 2)


def rate_with_method(fat, method):
    """Return raw steps, chosen steps and rate for round/int/floor.

    SNF is assumed to be the base value 8.5 so that only the fat-step
    difference is visible.
    """
    raw = (fat - BASE_FAT) / STEP_SIZE

    if method == "round":
        steps = round(raw)
    elif method == "int":
        steps = int(raw)
    elif method == "floor":
        steps = math.floor(raw)
    else:
        raise ValueError("Unknown method")

    rate = BASE_RATE + steps * FAT_STEP_VALUE
    return raw, steps, round(rate, 2)


def rate_round_only(fat):
    """Rate using round() for step count."""
    steps = round((fat - BASE_FAT) / STEP_SIZE)
    return round(BASE_RATE + steps * FAT_STEP_VALUE, 2)


def rate_int_only(fat):
    """Rate using int() truncation for step count."""
    steps = int((fat - BASE_FAT) / STEP_SIZE)
    return round(BASE_RATE + steps * FAT_STEP_VALUE, 2)


def rate_floor_only(fat):
    """Rate using math.floor() for step count."""
    steps = math.floor((fat - BASE_FAT) / STEP_SIZE)
    return round(BASE_RATE + steps * FAT_STEP_VALUE, 2)


def benchmark(func, iterations=200_000):
    """Measure total time and nanoseconds per call."""
    start = time.perf_counter()
    for _ in range(iterations):
        func(4.2)
    elapsed = time.perf_counter() - start
    return elapsed, elapsed / iterations * 1e9
