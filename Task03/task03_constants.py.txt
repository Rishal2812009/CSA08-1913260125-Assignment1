# ---------------------------------------------------------
# RATE CARD - single source of truth
# UPPER_CASE means: do not change these during a run.
# A rate revision is made HERE and nowhere else.
# ---------------------------------------------------------
BASE_RATE = 30.00        # Rs per litre at 3.5% fat and 8.5% SNF
BASE_FAT = 3.5           # reference fat percentage
BASE_SNF = 8.5           # reference SNF percentage

FAT_STEP_SIZE = 0.1      # percentage points per fat step
SNF_STEP_SIZE = 0.1      # percentage points per SNF step

FAT_STEP_VALUE = 0.40    # Rs per litre per fat step
SNF_STEP_VALUE = 0.30    # Rs per litre per SNF step

MIN_FAT = 2.0            # lowest allowed fat percentage
MAX_FAT = 7.0            # highest allowed fat percentage


def rate_per_litre(fat, snf):
    """Return the price per litre in Rs, rounded to 2 decimals.

    Parameters:
        fat (float): fat percentage of the milk sample.
        snf (float): solids-not-fat percentage of the sample.

    Returns:
        float: rate per litre, rounded to 2 decimal places.

    Reads the named constants. Never modifies them.
    """
    fat_steps = round((fat - BASE_FAT) / FAT_STEP_SIZE)
    snf_steps = round((snf - BASE_SNF) / SNF_STEP_SIZE)
    rate = BASE_RATE + fat_steps * FAT_STEP_VALUE + snf_steps * SNF_STEP_VALUE
    return round(rate, 2)


def payout(litres, fat, snf):
    """Return the amount payable in Rs for one entry, rounded to 2 decimals."""
    return round(litres * rate_per_litre(fat, snf), 2)
