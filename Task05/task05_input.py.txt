import time

def get_float_try_except(prompt):
    """Read a float from the user using try/except ValueError.
    
    This is the recommended approach because it handles all edge cases
    including leading/trailing spaces and scientific notation automatically.
    """
    while True:
        text = input(prompt)
        try:
            value = float(text)
            return value
        except ValueError:
            print(f"  Error: '{text}' is not a valid number.")
            print("  Please use digits and a dot (.) for decimals.")


def get_float_string_methods(prompt):
    """Alternative: Read a float by checking string methods first.
    
    This approach fails on valid Python floats like " 8.5 " or "-8.5" 
    unless heavily modified. It is shown here for comparison.
    """
    while True:
        text = input(prompt)
        
        # Remove commas if clerk typed "8,5" instead of "8.5"
        cleaned_text = text.replace(",", ".")
        
        # Check if it's a valid number format using string methods
        # Note: This simple check fails for negative numbers or spaces
        if cleaned_text.replace(".", "", 1).replace("-", "", 1).isdigit():
            return float(cleaned_text)
        else:
            print(f"  Error: String method rejected '{text}'.")
            print("  Please enter a valid number without letters.")


def get_positive_float(prompt):
    """Read a float that must be greater than zero."""
    while True:
        value = get_float_try_except(prompt)
        if value > 0:
            return value
        print("  Error: Value must be greater than zero.")


def get_valid_fat(prompt):
    """Read fat percentage between 2.0 and 7.0 inclusive."""
    while True:
        value = get_float_try_except(prompt)
        if 2.0 <= value <= 7.0:
            return value
        print("  Error: Fat must be between 2.0% and 7.0%.")
